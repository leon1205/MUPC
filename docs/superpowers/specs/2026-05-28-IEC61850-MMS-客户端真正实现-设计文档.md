## 1. 概述

### 1.1 背景

当前 `iec61850-plugin` 模块中的 MMS 客户端为简化实现，`encode_request` 仅生成 2 字节，不符合 IEC 61850 MMS 协议规范，无法与真实 IED 设备通信。

Phase 2 交付报告中明确标注：

> **注意**：IEC 61850 MMS 客户端封装不完整，实际部署需完善 MMS 协议栈。

### 1.2 目标

使用 **libIEC61850 C 库绑定**重构 MMS 客户端，提供符合 IEC 61850-7-2/7-3 标准的完整 MMS 协议支持。

### 1.3 范围

| 模块 | 功能 | 优先级 |
|------|------|--------|
| mms_client.rs | MMS 客户端封装（重构） | 必须 |
| mms_types.rs | MMS 数据类型定义（新建） | 必须 |
| asn1_utils.rs | ASN.1 编码/解码工具（新建） | 必须 |
| errors.rs | MMS 错误类型（更新） | 必须 |
| lib.rs | 模块导出（更新） | 必须 |

### 4.1 mms_types.rs - MMS 数据类型

```rust
//! MMS 数据类型定义
//!
//! 定义 IEC 61850-7-2 MMS 协议数据类型

/// MMS 数据对象
#[derive(Debug, Clone)]
pub struct DataObject {
    /// 逻辑节点名（如 "LLN0", "MMXU1"）
    pub ln: String,
    /// 数据对象名（如 "ST$Pos", "MX$Measriz`)
    pub do_name: String,
}

/// MMS 请求
#[derive(Debug, Clone)]
pub struct MmsRequest {
    pub service: MmsService,
    pub object: DataObject,
    pub payload: Vec<u8>,
}

/// MMS 服务类型
#[derive(Debug, Clone, Copy)]
pub enum MmsService {
    Read,
    Write,
    DefineVariableAccess,
    GetDataAccessAttributes,
}

/// MMS 响应
#[derive(Debug, Clone)]
pub struct MmsResponse {
    pub success: bool,
    pub data: Vec<u8>,
    pub error: Option<String>,
}
```

### 4.2 mms_client.rs - MMS 客户端封装

```rust
//! MMS 客户端封装
//!
//! 使用 libIEC61850 实现 MMS 协议客户端

use crate::config::MmsConfig;
use crate::errors::{Iec61850Error, Result};
use crate::mms_types::{MmsRequest, MmsResponse, DataObject, MmsService};
use std::sync::Arc;
use tokio::net::TcpStream;
use tokio::io::{AsyncReadExt, AsyncWriteExt};
use tokio::time::{timeout, Duration};

/// MMS 客户端状态
#[derive(Debug, Clone, PartialEq)]
pub enum MmsClientState {
    Disconnected,
    Connecting,
    Connected,
    Error(String),
}

/// MMS 客户端
pub struct MmsClient {
    config: MmsConfig,
    state: Arc<parking_lot::RwLock<MmsClientState>>,
}

impl MmsClient {
    /// 创建 MMS 客户端
    pub fn new(config: MmsConfig) -> Self {
        Self {
            config,
            state: Arc::new(parking_lot::RwLock::new(MmsClientState::Disconnected)),
        }
    }

    /// 连接到 IED（短连接模式）
    pub async fn connect(&self) -> Result<()> {
        let addr = format!("{}:{}", self.config.remote_ip, self.config.remote_port);

        // 短连接：每次请求建立新连接
        let stream = timeout(
            Duration::from_millis(self.config.connect_timeout_ms),
            TcpStream::connect(&addr),
        )
        .await
        .map_err(|_| Iec61850Error::MmsTimeout("连接超时".into()))?
        .map_err(|e| Iec61850Error::MmsConnectFailed(format!("连接失败: {}", e)))?;

        *self.state.write() = MmsClientState::Connected;
        Ok(())
    }

    /// 断开连接
    pub async fn disconnect(&self) -> Result<()> {
        *self.state.write() = MmsClientState::Disconnected;
        Ok(())
    }

    /// 获取客户端状态
    pub fn get_state(&self) -> MmsClientState {
        self.state.read().clone()
    }

    /// 发送 MMS 请求
    async fn send_request(&self, request: MmsRequest) -> Result<MmsResponse> {
        if self.get_state() != MmsClientState::Connected {
            return Err(Iec61850Error::MmsConnectFailed("未连接".into()));
        }

        // 使用 libIEC61850 编码请求
        let req_data = encode_mms_request(&request)?;

        // 发送请求
        let mut stream = TcpStream::connect(format!("{}:{}", self.config.remote_ip, self.config.remote_port)).await
            .map_err(|e| Iec61850Error::MmsConnectFailed(format!("连接失败: {}", e)))?;

        stream.write_all(&req_data).await
            .map_err(|e| Iec61850Error::ProtocolError(format!("发送失败: {}", e)))?;

        // 读取响应
        let mut resp_buf = vec![0u8; 8192];
        let n = timeout(
            Duration::from_millis(self.config.read_timeout_ms),
            stream.read(&mut resp_buf),
        )
        .await
        .map_err(|_| Iec61850Error::MmsTimeout("读取超时".into()))?
        .map_err(|e| Iec61850Error::ProtocolError(format!("读取失败: {}", e)))?;

        // 解码响应
        decode_mms_response(&resp_buf[..n])
    }

    /// 读取数据对象（Read 服务）
    pub async fn read_do(&self, ln: &str, do_name: &str) -> Result<Vec<u8>> {
        let request = MmsRequest {
            service: MmsService::Read,
            object: DataObject {
                ln: ln.to_string(),
                do_name: do_name.to_string(),
            },
            payload: Vec::new(),
        };

        let response = self.send_request(request).await?;

        if response.success {
            Ok(response.data)
        } else {
            Err(Iec61850Error::DataObjectNotFound(
                response.error.unwrap_or_default(),
            ))
        }
    }

    /// 写入数据对象（Write 服务）
    pub async fn write_do(&self, ln: &str, do_name: &str, value: &[u8]) -> Result<()> {
        let request = MmsRequest {
            service: MmsService::Write,
            object: DataObject {
                ln: ln.to_string(),
                do_name: do_name.to_string(),
            },
            payload: value.to_vec(),
        };

        let response = self.send_request(request).await?;

        if response.success {
            Ok(())
        } else {
            Err(Iec61850Error::WriteFailed(
                response.error.unwrap_or_default(),
            ))
        }
    }
}

/// MMS 客户端 trait（用于测试 mock）
#[async_trait::async_trait]
pub trait MmsClientTrait: Send + Sync {
    async fn connect(&self) -> Result<()>;
    async fn disconnect(&self) -> Result<()>;
    async fn read_do(&self, ln: &str, do_name: &str) -> Result<Vec<u8>>;
    async fn write_do(&self, ln: &str, do_name: &str, value: &[u8]) -> Result<()>;
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_mms_client_creation() {
        let config = MmsConfig::default();
        let client = MmsClient::new(config);
        assert_eq!(client.get_state(), MmsClientState::Disconnected);
    }
}
```

### 4.3 asn1_utils.rs - ASN.1 编码/解码

```rust
//! ASN.1 BER/DER 编码/解码工具
//!
//! 实现 IEC 61850 MMS 协议的 ASN.1 数据编码

use crate::errors::{Iec61850Error, Result};
use crate::mms_types::{MmsRequest, MmsResponse, DataObject, MmsService};

/// 编码 MMS 请求为 ASN.1 BER 格式
pub fn encode_mms_request(request: &MmsRequest) -> Result<Vec<u8>> {
    match request.service {
        MmsService::Read => encode_read_request(&request.object),
        MmsService::Write => encode_write_request(&request.object, &request.payload),
        MmsService::DefineVariableAccess => encode_define_request(&request.object),
        MmsService::GetDataAccessAttributes => encode_getda_request(&request.object),
    }
}

/// 解码 ASN.1 BER 响应
pub fn decode_mms_response(data: &[u8]) -> Result<MmsResponse> {
    if data.is_empty() {
        return Err(Iec61850Error::MmsInvalidResponse("空响应".into()));
    }

    // 检查 MMS PDU 类型
    match data[0] {
        0x81 => Ok(MmsResponse {
            success: true,
            data: data.to_vec(),
            error: None,
        }),
        0xA9 => Err(Iec61850Error::MmsProtocolError("Confirmed-ErrorPDU".into())),
        _ => Err(Iec61850Error::MmsInvalidResponse(format!(
            "未知响应类型: {:02x}",
            data[0]
        ))),
    }
}

/// 编码 Read 请求
fn encode_read_request(object: &DataObject) -> Result<Vec<u8>> {
    // 构建 MMS Read Request APDU (IEC 61850-8-1)
    let mut buf = Vec::new();

    buf.push(0x01); // confirmed-RequestPDU
```

```rust
#[derive(Debug, Clone)]
pub struct MmsTlsConfig {

    /// 客户端证书路径

}

impl Default for MmsTlsConfig {
    fn default() -> Self {
        Self {
            enabled: false,
            ca_cert_path: String::new(),
            client_cert_path: String::new(),
            client_key_path: String::new(),
            verify_peer: true,
        }
    }
}

/// MMS 配置
#[derive(Debug, Clone)]
pub struct MmsConfig {
    /// 本地 IP
```

### 6.1 错误类型

```rust
#[derive(Debug, thiserror::Error)]
pub enum Iec61850Error {
    #[error("MMS 连接失败: {0}")]
    MmsConnectFailed(String),

    #[error("MMS 请求超时: {0}")]
    MmsTimeout(String),

    #[error("MMS 协议错误: {0}")]
    MmsProtocolError(String),

    #[error("数据对象不存在: {0}")]
    DataObjectNotFound(String),

    #[error("写操作失败: {0}")]
    WriteFailed(String),

    #[error("无效响应: {0}")]
    MmsInvalidResponse(String),

    #[error("TLS 连接失败: {0}")]
    TlsConnectFailed(String),

    #[error("证书验证失败: {0}")]
    CertVerifyFailed(String),

    #[error("ASN.1 编码失败: {0}")]

    #[error("ASN.1 解码失败: {0}")]

}
```
