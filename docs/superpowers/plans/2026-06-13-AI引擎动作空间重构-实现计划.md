**Goal:** 将AI引擎动作空间从单参数(p_batt_set)重构为双参数(P_ref + k_droop)输出模式，实现时间尺度解耦和本质安全兜底。

```rust
/// 解析 RL 模型原始输出为 ActionOutput（双参数模式）
///
/// 输出格式: [p_ref, k_droop, load_shedding, pv_limit, confidence]
pub fn parse_action_output(raw: &[f32], config: &ActionSpaceConfig) -> Option<ActionOutput> {
    if raw.len() < 5 {
        return None;
    }

    // 获取 k_droop 范围（默认值为安全边界）
    let k_min = config.k_droop_min.unwrap_or(-100.0);
    let k_max = config.k_droop_max.unwrap_or(100.0);

    let mut action = ActionOutput {
        p_ref: (raw[0] as f64).clamp(
            -config.max_batt_discharge_power,
            config.max_batt_charge_power,
        ),
        k_droop: (raw[1] as f64).clamp(k_min, k_max),
        load_shedding: (raw[2] as f64).clamp(0.0, config.max_load_shedding),
        pv_limit: (raw[3] as f64).clamp(0.0, 1.0),
        confidence: raw.get(4).copied().unwrap_or(0.5) as f64,
    };

    action.confidence = action.confidence.clamp(0.0, 1.0);
    Some(action)
}

/// 解析 RL 模型原始输出为 ActionOutputLegacy（单参数模式，legacy）
```

```rust
    Dual,    // 双参数模式
    Legacy,  // 单参数模式（兼容）
}

impl ActionValidator {
    /// 创建 v2.0 模式校验器
    pub fn new_v2_0(config: ActionConstraintConfig) -> Self {
        Self {
            config,
            last_action: RwLock::new(None),
            droop_range: RwLock::new((-100.0, 100.0)), // 默认值
            mode: ActionSpaceMode::Dual,
        }
    }

    /// 创建 legacy 模式校验器
    pub fn new_legacy(config: ActionConstraintConfig) -> Self {
        Self {
            config,
            last_action: RwLock::new(None),
            droop_range: RwLock::new((-100.0, 100.0)),
            mode: ActionSpaceMode::Legacy,
        }
    }

    /// 更新 k_droop 范围（由 intercore 从实时控制模块获取后调用）

        *range = (k_min, k_max);

        *self.droop_range.read().unwrap()
    }
```

- [ ] **Step 2: 修改 validate 方法实现 ACT-DUAL-01~04**

```rust
/// ACT-DUAL-01: p_ref ∈ [-max_batt_discharge_power, max_batt_charge_power]
/// ACT-DUAL-02: k_droop ∈ [k_droop_min, k_droop_max]
/// ACT-DUAL-03: Δp_ref 变化率 <= p_batt_ramp_limit_kw / 步
/// ACT-DUAL-04: 当 dispatch_p_set 有效时，p_ref 绝对值不得超过 dispatch_p_set 绝对值
pub fn validate(
    &self,
    action: &ActionOutput,
    dispatch_p_set: Option<f64>,
    is_anti_reverse: bool,
    action_space_config: &ActionSpaceConfig,
) -> (ActionOutput, Vec<ViolationRecord>) {
    let mut validated = action.clone();
    let mut violations = Vec::new();
    let last = self.last_action.read().unwrap();

    // ACT-DUAL-01: p_ref 值域约束
    let p_ref_min = -action_space_config.max_batt_discharge_power;
    let p_ref_max = action_space_config.max_batt_charge_power;
    if validated.p_ref < p_ref_min {
        violations.push(ViolationRecord { rule: "ACT-DUAL-01", field: "p_ref", original: action.p_ref, clamped: p_ref_min });
        validated.p_ref = p_ref_min;
    } else if validated.p_ref > p_ref_max {
        violations.push(ViolationRecord { rule: "ACT-DUAL-01", field: "p_ref", original: action.p_ref, clamped: p_ref_max });
        validated.p_ref = p_ref_max;
    }

    // ACT-DUAL-02: k_droop 值域约束
    let (k_min, k_max) = *self.droop_range.read().unwrap();
    if validated.k_droop < k_min {
        violations.push(ViolationRecord { rule: "ACT-DUAL-02", field: "k_droop", original: action.k_droop, clamped: k_min });
        validated.k_droop = k_min;
    } else if validated.k_droop > k_max {
        violations.push(ViolationRecord { rule: "ACT-DUAL-02", field: "k_droop", original: action.k_droop, clamped: k_max });
        validated.k_droop = k_max;
    }

    // ACT-DUAL-03: p_ref 变化率约束
    if let Some(ref prev) = *last {
        let delta = (action.p_ref - prev.p_ref).abs();
        if delta > self.config.p_batt_ramp_limit_kw {
            let sign = if action.p_ref > prev.p_ref { 1.0 } else { -1.0 };
            validated.p_ref = prev.p_ref + sign * self.config.p_batt_ramp_limit_kw;
            violations.push(ViolationRecord { rule: "ACT-DUAL-03", field: "p_ref", original: action.p_ref, clamped: validated.p_ref });
        }
    }

    // ACT-DUAL-04: 调度指令权限约束
    if let Some(dp) = dispatch_p_set {
        if validated.p_ref.abs() > dp.abs() {
            let sign = validated.p_ref.signum();
            validated.p_ref = sign * dp.abs();
            violations.push(ViolationRecord { rule: "ACT-DUAL-04", field: "p_ref", original: action.p_ref, clamped: validated.p_ref });
        }
    }

    // load_shedding 和 pv_limit 保留，校验方式不变
    validated.load_shedding = validated.load_shedding.clamp(0.0, action_space_config.max_load_shedding);
    validated.pv_limit = validated.pv_limit.clamp(0.0, 1.0);
    validated.confidence = validated.confidence.clamp(0.0, 1.0);

    *self.last_action.write().unwrap() = Some(validated.clone());
    (validated, violations)
}
```

```rust
/// 核间通信帧 v2.0
///
/// v2.0 变更：
/// - ControlCmd 数据区携带 p_ref 和 k_droop 双参数
/// - FrameType 保持不变，通过数据区内容区分版本

/// 控制指令载荷 v2.0（JSON 格式）
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ControlCmdPayloadV2 {
    #[serde(rename = "p_ref")]
    pub p_ref: Option<f64>,
    #[serde(rename = "k_droop")]
    pub k_droop: Option<f64>,
    #[serde(rename = "load_shedding")]
    pub load_shedding: Option<f64>,
    #[serde(rename = "pv_limit")]
    pub pv_limit: Option<f64>,
    #[serde(rename = "ai_ready")]
    pub ai_ready: Option<bool>,
    #[serde(rename = "strategy_mode")]
    pub strategy_mode: Option<String>,
    #[serde(rename = "timestamp_ms")]
    pub timestamp_ms: Option<u64>,
    /// 帧版本号，用于区分 v1.x 和 v2.0
    #[serde(rename = "frame_version")]
    pub frame_version: Option<u8>,
}

impl ControlCmdPayloadV2 {
    pub const FRAME_VERSION: u8 = 2;

    pub fn new() -> Self {
        Self {
            p_ref: None,
            k_droop: None,
            load_shedding: None,
            pv_limit: None,
            ai_ready: None,
            strategy_mode: None,
            timestamp_ms: None,
            frame_version: Some(Self::FRAME_VERSION),
        }
    }

    pub fn from_action_output(action: &ActionOutput, ai_ready: bool, strategy_mode: &str) -> Self {
        Self {
            p_ref: Some(action.p_ref),
            k_droop: Some(action.k_droop),
            load_shedding: Some(action.load_shedding),
            pv_limit: Some(action.pv_limit),
            ai_ready: Some(ai_ready),
            strategy_mode: Some(strategy_mode.to_string()),
            timestamp_ms: Some(chrono::Utc::now().timestamp_millis() as u64),
            frame_version: Some(Self::FRAME_VERSION),
        }
    }

    /// 检测帧版本
    pub fn detect_version(data: &[u8]) -> Result<u8, MupcError> {
        match Self::from_json(data) {
            Ok(payload) => Ok(payload.frame_version.unwrap_or(1)),
            Err(_) => Ok(1), // 解析失败假设为 v1.x
        }
    }
}
```

```rust
/// 核间通信状态（用于通信中断检测）
#[derive(Debug, Clone)]
pub struct IntercoreConnectionState {
    /// 最后收到的有效 p_ref
    pub last_valid_p_ref: RwLock<Option<f64>>,
    /// 最后收到的有效 k_droop
    pub last_valid_k_droop: RwLock<Option<f64>>,

    pub last_heartbeat_ms: RwLock<u64>,
    /// 连接状态
    pub connected: RwLock<bool>,
}

impl IntercoreConnectionState {
    pub fn new() -> Self {
        Self {
            last_valid_p_ref: RwLock::new(None),
            last_valid_k_droop: RwLock::new(None),
            last_heartbeat_ms: RwLock::new(0),
            connected: RwLock::new(false),
        }
    }

    /// 更新收到的双参数
    pub fn update_valid_params(&self, p_ref: f64, k_droop: f64) {
        *self.last_valid_p_ref.write().unwrap() = Some(p_ref);
        *self.last_valid_k_droop.write().unwrap() = Some(k_droop);
    }

    /// 获取最后有效的双参数（通信中断时使用）
    pub fn get_last_valid_params(&self) -> (Option<f64>, Option<f64>) {
        let p_ref = *self.last_valid_p_ref.read().unwrap();
        let k_droop = *self.last_valid_k_droop.read().unwrap();
        (p_ref, k_droop)
    }

    /// 检查是否已收到有效参数
    pub fn has_valid_params(&self) -> bool {
        self.last_valid_p_ref.read().unwrap().is_some()
            && self.last_valid_k_droop.read().unwrap().is_some()
    }
```

```rust
impl AiIntegrator {
    /// 获取当前动作输出（支持降级）
    pub async fn get_action_output(&self) -> Result<ActionOutput, StrategyError> {
        let action = self.model_manager.full_decision_cycle().await?;

        // 检测通信中断
        if !self.is_intercore_connected().await {
            tracing::warn!("Intercore connection lost, activating fallback mode");
            *self.fallback_active.write().await = true;
            return self.get_fallback_action().await;
        }

        // 通信正常，更新最后有效参数
        self.intercore_state
            .update_valid_params(action.p_ref, action.k_droop);

        *self.fallback_active.write().await = false;
        Ok(action)
    }

    /// 获取降级动作（保持最后有效值）
    async fn get_fallback_action(&self) -> Result<ActionOutput, StrategyError> {
        let (p_ref, k_droop) = self.intercore_state.get_last_valid_params();

        match (p_ref, k_droop) {
            (Some(p), Some(k)) => {
                tracing::info!(
                    "Fallback: using last valid p_ref={}, k_droop={}",
                    p,
                    k
                );
                Ok(ActionOutput {
                    p_ref: p,
                    k_droop: k,
                    load_shedding: 0.0, // 降级时不修改负荷
                    pv_limit: 1.0,      // 降级时不限制光伏
                    confidence: 0.0,    // 降级模式置信度为 0
                })
            }
            _ => {
                // 无有效参数，使用安全默认值
                tracing::warn!("No valid params available, using safe defaults");
                Ok(ActionOutput {
                    p_ref: 0.0,
                    k_droop: 10.0, // 默认下垂系数
                    load_shedding: 0.0,
                    pv_limit: 1.0,
                    confidence: 0.0,
                })
            }
        }
    }

    /// 检查 intercore 连接状态
    async fn is_intercore_connected(&self) -> bool {
        *self.intercore_state.connected.read().unwrap()
    }
}
```
